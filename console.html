final class DisabledAccountHandler {
  private final SessionStore store;
  private final Counter revoked;
  private final AuditSink audit;
  private final DeadLetter deadLetter;
  private static final Logger log = Loggers.of(DisabledAccountHandler.class);

  void onRecord(Record<AccountDisabled> rec) {
    try {
      handle(rec.value());
    } catch (DecodeException | StoreException e) {
      deadLetter.send(rec, e);
    }
  }

  void handle(AccountDisabled ev) {
    List<String> ids = ev.liveSessionIds();
    int count;
    if (ids == null) {
      log.debug("skipped: no sessions field account={}", ev.accountId());
      return;
    } else if (ids.isEmpty()) {
      count = store.revokeAllForAccount(ev.tenantId(), ev.accountId());
    } else {
      count = 0;
      for (String id : ids) {
        if (store.revoke(ev.tenantId(), id, Reason.ACCOUNT_DISABLED)) {
          count++;
        }
      }
      store.revokeRefreshFamilies(ev.tenantId(), ids);
    }
    revoked.increment();
    audit.record(ev.accountId(), "sessions_revoked", count);
    log.debug("revoked sessions={} account={}", count, ev.accountId());
  }
}
